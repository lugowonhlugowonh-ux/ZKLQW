local function WorldToScreen(worldPos)
    local vp = Camera.ViewportSize
    local cf = Camera.CFrame
    local point = cf:PointToObjectSpace(worldPos)
    local fov = Camera.FieldOfView
    local aspect = vp.X / vp.Y

    local halfY = -point.Z * math.tan(math.rad(fov / 2))
    local halfX = aspect * halfY

    local screenX = (point.X / halfX) * 0.5 + 0.5
    local screenY = (point.Y / halfY) * 0.5 + 0.5

    local onScreen = -point.Z > 0 and screenX >= 0 and screenX <= 1 and screenY >= 0 and screenY <= 1

    return Vector2.new(screenX * vp.X, screenY * vp.Y), onScreen, -point.Z
end

local char_fly = {Enabled = false, Mode = "Hold", Speed = 38, Height = 7, Active = false, TargetY = nil}

local function ApplyCharFly()
    if not char_fly.Enabled or not char_fly.Active then return end

    local char = Player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    if not char_fly.TargetY then
        char_fly.TargetY = hrp.Position.Y + char_fly.Height
    end

    local diff = char_fly.TargetY - hrp.Position.Y
    if math.abs(diff) > 0.05 then
        hrp.Velocity = Vector3.new(hrp.Velocity.X, diff * 4, hrp.Velocity.Z)
    else
        hrp.Velocity = Vector3.new(hrp.Velocity.X, 0, hrp.Velocity.Z)
    end

    local moveDir = Vector3.zero
    if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + Camera.CFrame.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - Camera.CFrame.LookVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - Camera.CFrame.RightVector end
    if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + Camera.CFrame.RightVector end

    if moveDir.Magnitude > 0 then
        moveDir = moveDir.Unit
        hrp.Velocity = Vector3.new(moveDir.X * char_fly.Speed, hrp.Velocity.Y, moveDir.Z * char_fly.Speed)
    else
        hrp.Velocity = Vector3.new(0, hrp.Velocity.Y, 0)
    end

    hrp.AssemblyAngularVelocity = Vector3.zero
    hum.AutoRotate = true
end

    if not sa_whitelist_enabled then return true end

local function get_silent_target(fov_pos)
    local best, best_d = nil, math.huge
    local vp = Camera.ViewportSize
    local origin = silent_mode == "HP" and Vector2.new(vp.X / 2, vp.Y / 2) or fov_pos
    for _, p in pairs(Players:GetPlayers()) do
        if p == Player then continue end
        if not sa_is_allowed(p) then continue end
        local ch = p.Character
        local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
        local hum = ch and ch:FindFirstChildOfClass("Humanoid")
        local hit = ch and ch:FindFirstChild(silent_part)
        if not hrp or not hum or not hit or hum.Health <= 0 then continue end
        local sp, on_screen = WorldToScreen(hrp.Position)
        if not on_screen then continue end
        local d = (origin - sp).Magnitude
        if d < fov_radius and d < best_d then best = p; best_d = d end
    end
    return best
end

local function remove_esp_object(target_player)
    local obj = esp_objects[target_player]
    if not obj then return end
    for _, drawing in pairs(obj) do
        if drawing and drawing.Remove then drawing:Remove() end
    end
    esp_objects[target_player] = nil
end

local function update_esp()
    for target_player, obj in pairs(esp_objects) do
        if not target_player or not target_player.Parent then
            remove_esp_object(target_player)
            continue
        end
        local ch = target_player.Character
        local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
        local hum = ch and ch:FindFirstChildOfClass("Humanoid")
        local head = ch and ch:FindFirstChild("Head")
        local local_char = Player.Character
        local local_hrp = local_char and local_char:FindFirstChild("HumanoidRootPart")
        if not hrp or not hum or hum.Health <= 0 or not local_hrp then
            for _, drawing in pairs(obj) do drawing.Visible = false end
            continue
        end
        local dist = (hrp.Position - local_hrp.Position).Magnitude
        if dist > esp_max_distance then
            for _, drawing in pairs(obj) do drawing.Visible = false end
            continue
        end
        if esp_team_check and target_player.Team == Player.Team then
            for _, drawing in pairs(obj) do drawing.Visible = false end
            continue
        end
        local pos, on_screen = Camera:WorldToViewportPoint(hrp.Position)
        if not on_screen then
            for _, drawing in pairs(obj) do drawing.Visible = false end
            continue
        end
        local head_pos = Camera:WorldToViewportPoint(head.Position)
        local leg_pos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
        local box_height = math.abs(head_pos.Y - leg_pos.Y)
        local box_width = box_height * 0.55
        local color = esp_color
        if esp_boxes then
            obj.box.Size = Vector2.new(box_width, box_height)
            obj.box.Position = Vector2.new(pos.X - box_width / 2, pos.Y - box_height / 2)
            obj.box.Color = color
            obj.box.Visible = true
            obj.box_outline.Size = obj.box.Size
            obj.box_outline.Position = obj.box.Position
            obj.box_outline.Visible = true
        else
            obj.box.Visible = false
            obj.box_outline.Visible = false
        end
        if esp_names then
            obj.name.Position = Vector2.new(pos.X, pos.Y - box_height / 2 - 16)
            obj.name.Text = target_player.Name
            obj.name.Color = color
            obj.name.Visible = true
        else
            obj.name.Visible = false
        end
    end
end

local function UpdateESP3D()
    for player, data in pairs(esp3d.Objects) do
        if not player or not player.Parent then
            if data.Gui then data.Gui:Destroy() end
            esp3d.Objects[player] = nil
            continue
        end
        local char = player.Character
        if not char then
            if data.Gui then data.Gui.Adornee = nil end
            continue
        end
        if esp3d.TeamCheck and player.Team == Player.Team then
            data.Gui.Enabled = false
            continue
        end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum then
            if data.Gui then data.Gui.Adornee = nil end
            continue
        end
        local head = char:FindFirstChild("Head") or hrp
        data.Gui.Adornee = head
        local healthPct = hum.Health / hum.MaxHealth
        data.HealthFrame.Size = UDim2.new(healthPct, 0, 0.4, 0)
        data.HealthFrame.BackgroundColor3 = Color3.fromRGB(255 * (1 - healthPct), 255 * healthPct, 50)
        data.NameLabel.TextColor3 = player.Team == Player.Team and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 0, 0)
        local dist = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
        if dist then
            data.DistLabel.Text = math.floor((hrp.Position - dist.Position).Magnitude) .. "m"
        end
        data.NameLabel.Visible = esp3d.Name
        data.HealthFrame.Visible = esp3d.Health
        data.DistLabel.Visible = esp3d.Distance
        data.Frame.Visible = esp3d.Box
        data.Gui.Enabled = esp3d.Enabled
    end
end

    end
end

-- 🎯 FOV RENDER LOOP
-- ============================================================
RunService:BindToRenderStep("GhostFOV", 500, function()
    if not silent_fov_circle then return end

    local mouse_pos = UserInputService:GetMouseLocation()

    silent_fov_circle.Position = mouse_pos
    silent_fov_circle.Radius = fov_radius
    silent_fov_circle.Visible = show_fov and draw_fov

    -- Snapline
    if silent_aim and draw_snapline then
        local best_pos, best_d = nil, math.huge
        local origin = mouse_pos
        for _, p in pairs(Players:GetPlayers()) do
            if p == Player then continue end
            if not sa_is_allowed(p) then continue end
            local ch = p.Character
            local hum = ch and ch:FindFirstChildOfClass("Humanoid")
            local hit = ch and ch:FindFirstChild(silent_part)
            if not hit or not hum or hum.Health <= 0 then continue end
            local sp, on_screen = Camera:WorldToViewportPoint(hit.Position)
            if not on_screen then continue end
            local screenPos = Vector2.new(sp.X, sp.Y)
            local d = (origin - screenPos).Magnitude
            if d < fov_radius and d < best_d then
                best_d = d
                best_pos = screenPos
            end
        end
        if best_pos then
            silent_tracer_line.From = origin
            silent_tracer_line.To = best_pos
            silent_tracer_line.Visible = true
        else
            silent_tracer_line.Visible = false
        end
    else
        silent_tracer_line.Visible = false
    end
end)

-- ⌨️ KEYBIND HANDLERS
-- ============================================================
UserInputService.InputBegan:Connect(function(input)
    local keyName = input.KeyCode and input.KeyCode.Name
    if not keyName then return end

    -- Character Fly
    if keyName == "V" and char_fly.Enabled then
        if char_fly.Mode == "Toggle" then
            char_fly.Active = not char_fly.Active
            if char_fly.Active then
                char_fly.TargetY = nil
                print("✈️ FLY ON (Toggle)")
            else
                CharFlyLand()
                print("✈️ FLY OFF (Toggle)")
            end
        else
            char_fly.Active = true
            char_fly.TargetY = nil
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    local keyName = input.KeyCode and input.KeyCode.Name
    if keyName == "V" and char_fly.Enabled and char_fly.Mode == "Hold" then
        CharFlyLand()
    end
end)
